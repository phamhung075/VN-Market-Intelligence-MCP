**Change 2 — New exported pure function** (insert after `deriveKinhDichScore` ends at line 259):

```typescript
export function enrichDimensionScores(input: ConvictionInput): ConvictionInput {
  const enriched = { ...input };

  // newsSentimentScore [-1,1] → sentimentDirection + sentimentConfidence
  if (input.newsSentimentScore != null && enriched.sentimentDirection == null) {
    enriched.sentimentDirection =
      input.newsSentimentScore > 0.1 ? "bullish" :
      input.newsSentimentScore < -0.1 ? "bearish" : "neutral";
    enriched.sentimentConfidence = Math.abs(input.newsSentimentScore);
  }

  // kinhDichConfidenceRaw [0,100] → kinhDichScore [-1,+1]
  if (input.kinhDichConfidenceRaw != null && enriched.kinhDichScore == null) {
    // Normalize to [-1,+1]. Positive direction inferred from agentSignalsMajority if present.
    const sign = (input.agentSignalsMajority === "BUY") ? 1 :
                 (input.agentSignalsMajority === "SELL") ? -1 : 0;
    enriched.kinhDichScore = sign * (input.kinhDichConfidenceRaw / 100);
  }

  // agentSignalsMajority → cascadeDirection when no cascade data present
  if (input.agentSignalsMajority != null && enriched.cascadeDirection == null) {
    enriched.cascadeDirection =
      input.agentSignalsMajority === "BUY" ? "up" :
      input.agentSignalsMajority === "SELL" ? "down" : "neutral";
    enriched.cascadeConfidence = 0.6; // moderate confidence for agent vote signal
  }

  return enriched;
}
```

```typescript
/**
 * Format a 6-dimension conviction breakdown as a Vietnamese narrative block.
 * Result is NOT passed through TelegramMessageFactory (no truncation).

export function formatConvictionBlock(result: ConvictionResult): string {
  const d = result.dimensions;
  const pct = (n: number) => `${Math.round(n * 100)}%`;
  return [
    `Tại sao: ${result.summary}`,
    `Xác nhận: Giá ${pct(d.priceAction)} | KL ${pct(d.volumeConfirmation)} | Ngành ${pct(d.sectorAlignment)}`,
    `Kinh Dịch: ${pct(d.kinhDich)} tin cậy`,
    `Tiếp theo: Tín hiệu vĩ mô ${pct(d.cascade)} | Cảm xúc thị trường ${pct(d.sentiment)}`,
    `Rủi ro: [caller must pass complete risk list — no truncation]`,
  ].join("\n");
}
```
