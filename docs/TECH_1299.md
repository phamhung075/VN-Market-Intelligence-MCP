## Overview

Reduce MCP tool context from 65k → <30k tokens by implementing:
1. **Tool Index** — 1-liner reference for all 106 tools (no functional change)
2. **Skill-Gated Bootstrap** — Filter tool list by declared skills (core logic)
3. **Session Cache** — Track tool usage for future optimization (analytics)

No new domain models, no external APIs, no schema changes. Pure refactoring of bootstrap logic + cron job.

```typescript
import { SKILL_MANIFEST, ALWAYS_ON_TOOLS } from '../../domain/skillManifest';
import { sessionToolCache } from '../../infrastructure/cache/sessionToolCache';

export interface AgentBootstrapOptions {
  sessionId: string;
  skillIds?: string[]; // NEW: optional array of skill names
}

export async function getAgentContext(
  options: AgentBootstrapOptions
): Promise<{ context: string; toolCount: number }> {
  const { sessionId, skillIds } = options;

  // Try session cache first
  const cached = sessionToolCache.get(sessionId);
  if (cached) {
    // Cache hit: return cached tools (same session within 8h TTL)
    return {
      context: await formatToolsAsContext(cached.tools),
      toolCount: cached.tools.length,
    };
  }

  let tools: string[];

  if (!skillIds || skillIds.length === 0) {
    // LEGACY MODE: Load all tools (backwards compatibility)
    console.warn(
      `[bootstrap] No skillIds provided. Loading all 106 tools (legacy mode).`
    );
    tools = await mcp.getAllTools();
  } else {
    // NEW MODE: Compute skill-union
    tools = computeSkillUnion(skillIds);

    // Validate: ensure all tools exist in MCP registry
    const allMcpTools = await mcp.getAllToolNames();
    const invalid = tools.filter((t) => !allMcpTools.includes(t));
    if (invalid.length > 0) {
      logger.warn(
        `[bootstrap] Invalid tools in manifest: ${invalid.join(', ')}`
      );
      tools = tools.filter((t) => allMcpTools.includes(t));
    }
  }

  // Cache the tool list for this session
  sessionToolCache.set(sessionId, tools);

  return {
    context: await formatToolsAsContext(tools),
    toolCount: tools.length,
  };
}

/**
 * Compute tool union from skill IDs.
 * Returns all tools required by any skill + always-on tools.
 */
function computeSkillUnion(skillIds: string[]): string[] {
  const toolSet = new Set<string>();

  // Add always-on tools
  ALWAYS_ON_TOOLS.forEach((tool) => toolSet.add(tool));

  // Add skill-specific tools
  skillIds.forEach((skillId) => {
    const skillTools = SKILL_MANIFEST[skillId];
    if (!skillTools) {
      console.warn(
        `[bootstrap] Skill not found in manifest: ${skillId}. Skipping.`
      );
      return;
    }
    skillTools.forEach((tool) => toolSet.add(tool));
  });

  return Array.from(toolSet).sort();
}
```

### 3. Session Cache (Code)

#### File: `src/infrastructure/cache/sessionToolCache.ts` (New)

```typescript
import NodeCache from 'node-cache';

interface SessionToolEntry {
  tools: string[];
  lastAccess: Date;
}

/**
 * In-memory LRU cache for session → tools mapping.
 * - TTL: 8 hours (typical user session duration)
 * - Max: 100 sessions (~1MB memory)
 * - Use case: Store tools loaded per session to enable analytics
 */
class SessionToolCache {
  private cache: NodeCache;
  private sessionMap: Map<string, SessionToolEntry> = new Map();
  private readonly MAX_SESSIONS = 100;

  constructor() {
    // node-cache: TTL 8h = 28800 seconds
    this.cache = new NodeCache({ stdTTL: 28800, checkperiod: 600 });
    this.cache.on('del', (key) => {
      this.sessionMap.delete(key);
    });
  }

  get(sessionId: string): SessionToolEntry | undefined {
    const entry = this.cache.get<SessionToolEntry>(sessionId);
    if (entry) {
      entry.lastAccess = new Date();
    }
    return entry;
  }

  set(sessionId: string, tools: string[]): void {
    // LRU eviction: if at capacity, remove oldest entry
    if (this.sessionMap.size >= this.MAX_SESSIONS) {
      const oldest = Array.from(this.sessionMap.entries()).sort(
        (a, b) => a[1].lastAccess.getTime() - b[1].lastAccess.getTime()
      )[0];
      if (oldest) {
        this.cache.del(oldest[0]);
        this.sessionMap.delete(oldest[0]);
      }
    }

    const entry: SessionToolEntry = {
      tools,
      lastAccess: new Date(),
    };
    this.cache.set(sessionId, entry);
    this.sessionMap.set(sessionId, entry);
  }

  clear(): void {
    this.cache.flushAll();
    this.sessionMap.clear();
  }

  getAllSessions(): Map<string, SessionToolEntry> {
    return new Map(this.sessionMap);
  }
}

export const sessionToolCache = new SessionToolCache();
```

### 4. Cron Job for Tool Usage Analytics

#### File: `src/infrastructure/scheduler/trackSessionToolUsageJob.ts` (New)

```typescript
import { sessionToolCache } from '../cache/sessionToolCache';
import { writeFileSync } from 'fs';
import { join } from 'path';

interface ToolUsageStats {
  timestamp: Date;
  period: string; // "last_7_days"
  tools: Record<string, { count: number; percentage: number }>;
  sessionCount: number;
  uniqueToolsUsed: number;
  averageToolsPerSession: number;
}

/**
 * Cron job: Run every 8 hours
 * Purpose: Aggregate tool usage across all sessions, generate histogram
 * Output: docs/agent-memory/modules/tool-usage-stats.json
 */
export async function trackSessionToolUsageJob(): Promise<void> {
  const startTime = Date.now();

  try {
    // Aggregate all sessions
    const sessions = sessionToolCache.getAllSessions();
    const toolFrequency: Record<string, number> = {};
    let totalToolCalls = 0;

    sessions.forEach(({ tools }) => {
      tools.forEach((tool) => {
        toolFrequency[tool] = (toolFrequency[tool] || 0) + 1;
        totalToolCalls++;
      });
    });

    // Compute statistics
    const uniqueTools = Object.keys(toolFrequency).length;
    const stats: ToolUsageStats = {
      timestamp: new Date(),
      period: "last_7_days",
      tools: {},
      sessionCount: sessions.size,
      uniqueToolsUsed: uniqueTools,
      averageToolsPerSession:
        sessions.size > 0 ? totalToolCalls / sessions.size : 0,
    };

    // Build tool histogram with percentages
    Object.entries(toolFrequency).forEach(([tool, count]) => {
      stats.tools[tool] = {
        count,
        percentage:
          totalToolCalls > 0 ? (count / totalToolCalls) * 100 : 0,
      };
    });

    // Sort by frequency (descending)
    const sorted = Object.entries(stats.tools)
      .sort((a, b) => b[1].count - a[1].count)
      .reduce((acc, [tool, data]) => {
        acc[tool] = data;
        return acc;
      }, {} as Record<string, { count: number; percentage: number }>);

    stats.tools = sorted;

    // Write to agent-memory
    const outputPath = join(
      process.cwd(),
      "docs/agent-memory/modules/tool-usage-stats.json"
    );
    writeFileSync(outputPath, JSON.stringify(stats, null, 2));

    console.log(
      `[trackSessionToolUsageJob] Aggregated ${sessions.size} sessions, ` +
      `${uniqueTools} unique tools, ${totalToolCalls} total tool loads`
    );

    // Log completion
    await logSchedulerCompletion(
      "trackSessionToolUsageJob",
      Date.now() - startTime,
      "ok"
    );
  } catch (error) {
    console.error("[trackSessionToolUsageJob] Error:", error);
    await logSchedulerCompletion(
      "trackSessionToolUsageJob",
      Date.now() - startTime,
      "error"
    );
    throw error;
  }
}
```

## Decision Log

**Q: Why not use Redis/Memcached for session cache?**
A: Overkill for 100 sessions (~1MB memory). In-memory LRU is simpler, zero-infra, TTL handles expiry.

**Q: Why SKILL_MANIFEST as static const, not database table?**
A: Manifest changes rarely (when new tool added/skill defined). Static const + code review is safer than runtime config drift. Future: can move to JSON config if needed.

**Q: Why 8-hour TTL for sessions?**
A: Market hours = ~8–10h (9:15am–4:30pm VN time). Cache expires between sessions. Prevents stale tool lists.

**Q: Why not drop unused tools entirely?**
A: Tools are discoverable by agents in edge cases. Filtering is safer than removal. Usage stats will inform future deprecation.

**Q: Why skill-union instead of per-agent roles?**
A: Agents can multi-task (news scout + market watcher in same session). Union handles flexibility. Per-agent roles are future optimization.
